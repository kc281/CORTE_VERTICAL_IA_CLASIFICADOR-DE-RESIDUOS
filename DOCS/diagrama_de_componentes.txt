actor "Usuario\n(empleado / supervisor /\nadministrador)" as Usuario

' ───────────────────────── Frontend ─────────────────────────
node "Navegador web\n(desplegado en Vercel)" as Navegador {
  package "Frontend React + Vite + Tailwind" as Frontend {
    component "App" as App
    component "AuthScreen" as AuthScreen
    component "PrivacyModal" as PrivacyModal
    component "ResultCard" as ResultCard
    component "HistoryTable" as HistoryTable
    component "StatsDashboard\n(Recharts)" as StatsDashboard
    component "api.js\n(cliente HTTP Axios)" as ApiClient
    component "supabaseClient.js" as SupabaseClient
  }
}

' ───────────────────────── Backend ──────────────────────────
node "Servidor backend\n(Docker en Render)" as Servidor {
  package "Backend FastAPI" as Backend {

    package "Capa de API (controladores)" as CapaAPI {
      component "routes.py" as Routes
      component "auth.py\n(validación JWT)" as Auth
      component "utils.py\n(validación de imagen)" as Utils
    }

    package "Capa de servicios de aplicación" as CapaServicios {
      component "ClasificacionService\n[RF-01 + escritura RF-02]" as ClasifSvc
      component "HistorialService\n[RF-02]" as HistSvc
      component "EstadisticasService\n[RF-03]" as StatsSvc
      component "EmpresaService\n[soporte]" as EmpSvc
    }

    package "Capa de persistencia" as CapaPersistencia {
      component "PredictionRepository\n(crud.py)" as Repo
      component "Modelos ORM\n(models.py)" as ORM
      component "database.py\n(engine / sesión)" as DB
    }

    package "Componente de IA" as CapaIA {
      component "model_loader.py\n(preprocesamiento + inferencia)" as ModelLoader
      artifact "mobilenetv2_waste.h5\n(TensorFlow / Keras)" as Modelo
    }
  }
}

' ───────────────────────── Servicios externos ───────────────
cloud "Supabase" as Supabase {
  component "Supabase Auth\n(emite JWT, publica JWKS)" as SupaAuth
  database "PostgreSQL\nprediction_logs\ncompanies\ncompany_members" as Postgres
}

' ───────────────────────── Interfaz del API ─────────────────
interface "API REST\n(HTTPS + JSON)" as IApi
Routes -up- IApi

note right of IApi
  <b>Endpoints expuestos por routes.py</b>
  |= RF |= Método y ruta |= Servicio |
  | RF-01 | POST /predict | ClasificacionService |
  | RF-02 | GET /history | HistorialService |
  | RF-02 | GET /predict/{id} | HistorialService |
  | RF-02 | PATCH /predict/{id}/verify | HistorialService |
  | RF-03 | GET /stats | EstadisticasService |
  | Soporte | POST /register-company | EmpresaService |
  | Soporte | GET /company-info | EmpresaService |
end note

' ───────────────────────── Relaciones ───────────────────────
Usuario --> App : usa (HTTPS)

App --> AuthScreen
App --> PrivacyModal
App --> ResultCard
App --> HistoryTable
App --> StatsDashboard
App --> ApiClient
HistoryTable --> ApiClient
StatsDashboard --> ApiClient
AuthScreen --> SupabaseClient
ApiClient --> SupabaseClient : obtiene\naccess_token

SupabaseClient --> SupaAuth : inicio de sesión / registro

ApiClient ..> IApi : Bearer JWT

Routes --> Auth : valida token
Auth --> SupaAuth : descarga JWKS
Routes --> Utils : formato y tamaño
Routes --> ClasifSvc
Routes --> HistSvc
Routes --> StatsSvc
Routes --> EmpSvc

ClasifSvc --> ModelLoader : clasifica imagen
ModelLoader --> Modelo : carga
ClasifSvc --> Repo
HistSvc --> Repo
StatsSvc --> Repo
EmpSvc --> Repo

Repo --> ORM
Repo --> DB
DB --> Postgres : SQLAlchemy + psycopg\n(SSL)

note right of ClasifSvc
  Reglas de negocio:
  - Advertencia si confianza < 0.60
  - Nunca guarda la imagen (Ley 1581 de 2012)
  - Responde 201 solo tras confirmar el guardado
end note

note bottom of Postgres
  Aislamiento multiempresa:
  todas las consultas se filtran por company_id
end note
